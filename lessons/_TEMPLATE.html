# Gọi Messages API bằng Python SDK
import anthropic
client = anthropic.Anthropic()  # đọc ANTHROPIC_API_KEY từ biến môi trường
msg = client.messages.create(
    model="MODEL_ID_TU_TAI_LIEU",
    max_tokens=1024,
    messages=[{"role": "user", "content": "Xin chào Claude"}],
)
print(msg.content[0].text)
